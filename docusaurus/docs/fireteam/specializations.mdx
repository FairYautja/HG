# Fireteam Specializations

Passive abilities that provide buffs to enhance specific playstyles.
Each class has access to a different set of specializations.


## Stats

export const modify = {
    Spec_Overseer: { callback: (value) => `+${value} HP`, good: true },
    Spec_Liberator: { callback: (value) => `+${value} HP`, good: true },
    Spec_Scavenger: { callback: (value) => `+${value} HP`, good: true },
    Spec_Spotter: { callback: (value) => `${value} °C/${(value*9/5)+32} °F`, good: true },
    Spec_FieldMedic: { good_negative: true },
    Spec_Comms: { callback: (value) => `${value} s faster`, good: true },
}

export const modify_secondary = {
    Spec_Spotter: { good_negative: true },
}
