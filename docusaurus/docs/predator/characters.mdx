export function fmtArmor(key) {
    return function format(row) {
        const v = row.armor?.[key];
        if (v == null) return null;
        if (v === 1.0) return <span style={{color: 'var(--ifm-color-secondary)'}}>—</span>;
        const pct = Math.round((v - 1) * 100);
        const color = pct > 0 ? '#f0001f' : '#218409';
        return <span style={{fontWeight: 'bold', color}} title={`x${v}`}>{pct > 0 ? '+' : ''}{pct}%</span>;
    };
}

export function fmtBeam(row) {
    const v = row.beam_switch;
    if (v == null) return <span style={{color: 'var(--ifm-color-secondary)'}}>default</span>;
    return <span>{Math.round(v / 100)} <small>m</small></span>;
}

## Resistances & Plasma Caster Range

:::info
Resistance values show how much incoming damage is **modified** for each damage type. Negative
(green) means the Predator takes less damage; positive (red) means more. Blank means the same
as the default (no special resistance). **Beam switch** is the maximum distance at which the
Plasma Caster targeting beam can jump between targets.
:::

## Notes

:::info[Second Wind bleed]
  All Predators bleed 5 HP per second when they are in Second Wind. Theoretically, this gives them
  between 3:20 and 5:00 minutes to heal themselves. However, they take additional damage if
  the Fireteam or NPCs shoot at them.
:::
