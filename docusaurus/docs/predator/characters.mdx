export const CLOAK_REGEN_DELAY = 4.2;

export const predators = data.characters.predator.map((c) => ({
    ...c,
    beam_switch_m: c.beam_switch != null ? c.beam_switch / 100 : null,
    cloak_time: cloakTime(c),
    stamina_refill: ratio(c.stamina, c.stamina_recovery),
    energy_refill: ratio(c.energy, c.energy_recovery),
}));

export function cloakTime(c) {
    // Seconds cloaked from full energy. The activation cost is paid once, the drain runs per
    // second, and energy regeneration resumes CLOAK_REGEN_DELAY seconds after cloaking. The delay
    // isn't in the game files; it's estimated from an in-game test (Jotun uncloaks after ~12 s,
    // Celtic never does).
    const cloak = c.cloak;
    if (!cloak || c.energy == null || c.energy_recovery == null) return null;
    const available = c.energy - cloak.initial_energy;
    if (available <= cloak.energy * CLOAK_REGEN_DELAY) {
        return Math.round(available / cloak.energy * 10) / 10;  // runs out before regen starts
    }
    const net = cloak.energy - c.energy_recovery;
    if (net <= 0) return Infinity;
    return Math.round((available - c.energy_recovery * CLOAK_REGEN_DELAY) / net * 10) / 10;
}

export const cloakTimes = predators.map((c) => c.cloak_time).filter((v) => v != null && isFinite(v));
export const cloakMax = Math.max(...cloakTimes);
export const cloakMin = Math.min(...cloakTimes);

export function fmtCloakTime(row) {
    const cloak = row.cloak;
    const t = row.cloak_time;
    if (!cloak || t == null) return null;
    const title = `(${row.energy} energy - ${cloak.initial_energy} activation - ${row.energy_recovery} recovery × ${CLOAK_REGEN_DELAY} s delay) / `
        + `(${cloak.energy} drain - ${row.energy_recovery} recovery) · ${cloak.id}`;
    if (!isFinite(t)) {
        return <span title={title + ' · recovery ≥ drain'}><strong>∞</strong></span>;
    }
    const percent = cloakMax === cloakMin ? 100 : Math.round((t - cloakMin) / (cloakMax - cloakMin) * 100);
    return (
        <div title={title}>
            <strong>{t} s</strong><br />
            <progress value={percent} max="100" style={{ width: '60px' }}>{percent}%</progress>
        </div>
    );
}

*Show details* adds refill times, stamina exhaustion, cloak time and beam switch distance. *Stamina
refill* and *Energy refill* are the seconds needed to recover from empty to full (hover for the
raw recovery rate).
*Cloak time* is how long the Predator can stay cloaked starting from full energy, standing still.
Cloaking costs the activation energy once, then drains energy every second. Energy keeps
regenerating while cloaked, but only after a delay of about 4.2 s, so for the first seconds the full
drain applies. `∞` means energy recovery is at least as fast as the drain. Verified in game: Celtic
stays cloaked indefinitely, Jotun uncloaks after about 12 s. The 4.2 s delay is estimated from the
Jotun test; it isn't part of the game files. Hover a value to see the numbers used.


## Effective Health

Effective health shows how much raw weapon damage each Predator can absorb per damage type before
entering Second Wind: base HP ÷ armor. Armor multiplies incoming damage, which was verified in game:
4–5 D34-D grenades (15 direct + 120 splash, ×2.0 launcher armor) put a 1,000 HP Scout into
Second Wind. If armor divided damage instead, it would take 14 or more.

### Headshots

:::note
Weapons can add their own headshot bonus on top of the body part modifiers above. See the
*Headshot* column in [Fireteam weapons](../fireteam/weapons.mdx#stats) and
[Predator weapons](../predator/weapons.mdx#stats).

Predator weapons with both a melee attack and a throw (e.g. Combistick, Trident, Smart Disc) can
have a different headshot bonus for each. Those show two values as *melee / throw*, so `+5% / +20%`
means +5% for melee hits and +20% for the thrown weapon. `—` marks the attack without a bonus.
:::

export const secondWind = (() => {
    const hp = pd.map((c) => c.downed_health).filter((v) => v != null);
    const time = (s) => `${Math.floor(s / 60)}:${String(Math.round(s % 60)).padStart(2, '0')}`;
    return { min: time(Math.min(...hp) / downedBleed), max: time(Math.max(...hp) / downedBleed) };
})();

:::info[Second Wind bleed]
  All Predators bleed {downedBleed} HP per second when they are in Second Wind. Theoretically, this
  gives them between {secondWind.min} and {secondWind.max} minutes to heal themselves. However, they
  take additional damage if the Fireteam or NPCs shoot at them.
:::
